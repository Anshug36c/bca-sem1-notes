1. Temperature: Celsius → Fahrenheit 🌡️

#include <stdio.h>
int main() {
    float c, f;
    printf("Enter temperature in Celsius: ");
    scanf("%f", &c);
    f = (c * 9 / 5) + 32;          // formula: F = C×9/5 + 32
    printf("%.2f Celsius = %.2f Fahrenheit", c, f);
    return 0;
}
Input: 38  →  Output: 38.00 Celsius = 100.40 Fahrenheit
(Aur idea: 0°C = 32°F, 100°C = 212°F — formula ke andar yahi khayal hai)

2. Simple Interest & Compound Interest 💰

#include <stdio.h>
#include <math.h>
int main() {
    float p, r, t, si, ci;
    printf("Enter Principal, Rate, Time: ");
    scanf("%f %f %f", &p, &r, &t);

    si = (p * r * t) / 100;                        // SI = P×R×T/100
    ci = p * pow((1 + r / 100), t) - p;            // CI = P(1+R/100)^T − P

    printf("Simple Interest  = %.2f\n", si);
    printf("Compound Interest = %.2f\n", ci);
    return 0;
}
Input: 10000 5 2
Simple Interest  = 1000.00
Compound Interest = 1025.00
(Kyun CI zyada? Kyunki CI me "interest pe bhi interest" milta hai 😄)
Real life: Bank FD par SI, loan/bachat schemes me CI — "5% interest 2 saal" ka matlab depend karta hai compounding pe. Isi liye SIP/compound me paisa tez badhta hai. 💹

4. Swap Two Numbers — with third variable 🔄

#include <stdio.h>
int main() {
    int a, b, temp;
    scanf("%d %d", &a, &b);
    printf("Before: a=%d b=%d\n", a, b);

    temp = a;   // a ko temp me rakha
    a = b;      // b ki value a me
    b = temp;   // temp (purana a) b me

    printf("After : a=%d b=%d", a, b);
    return 0;
}
Input: 10 20 → After: a=20 b=10
(Glass analogy: do glass ka paani exchange karna — ek khaali glass (temp) chahiye!)

8. Roots of Quadratic Equation (ax² + bx + c = 0) 🧮

#include <stdio.h>
#include <math.h>
int main() {
    float a, b, c, d, r1, r2;
    printf("Enter a, b, c: ");
    scanf("%f %f %f", &a, &b, &c);

    d = b * b - 4 * a * c;      // discriminant

    if (d > 0) {
        r1 = (-b + sqrt(d)) / (2 * a);
        r2 = (-b - sqrt(d)) / (2 * a);
        printf("Real & distinct roots: %.2f and %.2f", r1, r2);
    } else if (d == 0) {
        r1 = -b / (2 * a);
        printf("Real & equal roots: %.2f", r1);
    } else {
        printf("Complex roots: %.2f + %.2fi and %.2f - %.2fi",
               -b/(2*a), sqrt(-d)/(2*a), -b/(2*a), sqrt(-d)/(2*a));
    }
    return 0;
}
Input: 1 -5 6 → Real & distinct roots: 3.00 and 2.00
(d > 0: do roots, d = 0: ek root, d < 0: imaginary roots — "i" wale)

9. Armstrong Number 💪

#include <stdio.h>
int main() {
    int n, temp, r, sum = 0;
    scanf("%d", &n);
    temp = n;
    while (temp != 0) {
        r = temp % 10;              // digit nikala
        sum += r * r * r;           // cube joda
        temp /= 10;
    }
    if (sum == n)
        printf("%d is Armstrong", n);
    else
        printf("%d is not Armstrong", n);
    return 0;
}
153 = 1³+5³+3³ = 1+125+27 = 153 ✔ Armstrong
370, 371, 407 bhi Armstrong hain (3-digit). 1634 bhi hota hai (4-digit, 4th power se!)

    int a, b, c;

    else
        printf("%d is largest", c);

}

P4. Roots of Quadratic Equation using SWITCH 🔀

#include <stdio.h>
#include <math.h>
int main() {
    float a, b, c, d, r1, r2;
    scanf("%f %f %f", &a, &b, &c);
    d = b * b - 4 * a * c;

    // d ki condition ko "1, 0, -1" me convert karo taaki switch chal sake
    if (d > 0) d = 1;
    else if (d == 0) d = 0;
    else d = -1;

    switch ((int)d) {
        case 1:
            r1 = (-b + sqrt(b*b - 4*a*c)) / (2*a);
            r2 = (-b - sqrt(b*b - 4*a*c)) / (2*a);
            printf("Real & distinct roots: %.2f and %.2f", r1, r2);
            break;
        case 0:
            printf("Real & equal roots: %.2f", -b / (2*a));
            break;
        case -1:
            printf("Complex roots: %.2f ± %.2fi",
                   -b/(2*a), sqrt(-(b*b - 4*a*c))/(2*a));
            break;
    }
    return 0;
}
Input: 1 -5 6 → Real & distinct roots: 3.00 and 2.00
Note: switch me case "range" nahi ho sakta (d>0/d=0 direct nahi) — isliye pehle 
condition ko 1/0/−1 me convert kiya. Ye trick examiner ko dikhana! ✔

Isme kya sikha: switch sirf equality pe chalta hai, isliye range conditions ko "code" me convert karke switch me bhejna — realistic programming skill.

P7. Product of Two Matrices ✖️

#include <stdio.h>
int main() {
    int a[10][10], b[10][10], prod[10][10];
    int r1, c1, r2, c2, i, j, k;

    printf("Rows & cols of A: ");
    scanf("%d %d", &r1, &c1);
    printf("Rows & cols of B: ");
    scanf("%d %d", &r2, &c2);

    if (c1 != r2) {                       // condition check!
        printf("Multiplication not possible (c1 != r2)");
        return 0;
    }

    printf("Enter matrix A:\n");
    for (i = 0; i < r1; i++)
        for (j = 0; j < c1; j++)
            scanf("%d", &a[i][j]);

    printf("Enter matrix B:\n");
    for (i = 0; i < r2; i++)
        for (j = 0; j < c2; j++)
            scanf("%d", &b[i][j]);

    for (i = 0; i < r1; i++) {
        for (j = 0; j < c2; j++) {
            prod[i][j] = 0;
            for (k = 0; k < c1; k++) {            // dot product
                prod[i][j] += a[i][k] * b[k][j];
            }
        }
    }

    printf("Product:\n");
    for (i = 0; i < r1; i++) {
        for (j = 0; j < c2; j++)
            printf("%d ", prod[i][j]);
        printf("\n");
    }
    return 0;
}
A = [1 2; 3 4], B = [5 6; 7 8]

         43 50       (Maths Unit 1 wala hi kaam, ab C me! 😄)

P8. GCD of Two Numbers — Iteration aur Recursion dono 🌀

/* (a) Iteration se (Euclid's algorithm) */
#include <stdio.h>
int main() {
    int a, b, t;
    scanf("%d %d", &a, &b);
    while (b != 0) {
        t = b;
        b = a % b;      // remainder
        a = t;
    }
    printf("GCD = %d", a);
    return 0;
}
Input: 36 60 → GCD = 12

/* (b) Recursion se */
int gcd(int a, int b) {
    if (b == 0)
        return a;                  // base case
    return gcd(b, a % b);          // khud ko chhote input se call
}
gcd(36, 60) → gcd(60, 36) → gcd(36, 24) → gcd(24, 12) → gcd(12, 0) → 12 ✔

Isme kya sikha: Euclid's algorithm; aur ek hi problem ke do solutions — iteration (loop) vs recursion. Viva me poochhte hain: "recursion me base case ka kya role hai?"

P9. Storage Classes Illustration 🗄️

#include <stdio.h>

void demo() {

P10. Call by Value vs Call by Reference 🔄

#include <stdio.h>

void swapValue(int x, int y) {         // copies aayi
    int t = x;  x = y;  y = t;
    printf("Andar (value):     x=%d y=%d\n", x, y);
}

void swapRef(int *x, int *y) {         // addresses aaye
    int t = *x;  *x = *y;  *y = t;
    printf("Andar (reference): x=%d y=%d\n", *x, *y);
}

int main() {
    int a = 10, b = 20;

    swapValue(a, b);
    printf("main me (baad value):     a=%d b=%d\n", a, b);   // 10 20 — no change!

    swapRef(&a, &b);
    printf("main me (baad reference): a=%d b=%d\n", a, b);   // 20 10 — changed ✔
    return 0;
}
OUTPUT:
Andar (value):     x=20 y=10
main me (baad value):     a=10 b=20    ← original safe!
Andar (reference): x=20 y=10
main me (baad reference): a=20 b=10    ← original badal gaya ✔

Isme kya sikha: value = photocopy, reference = original ka pata. Pointer ka practical use — yahi concept function se "multiple values wapas laane" me kaam aata hai.

P11. Count Occurrences of Each Alphabet (command line arguments) 🔤

#include <stdio.h>
#include <ctype.h>

int main(int argc, char *argv[]) {
    int count[26] = {0};
    int i, j;

    // har argument ke har character ko check karo
    for (i = 1; i < argc; i++) {
        for (j = 0; argv[i][j] != '\0'; j++) {
            char ch = tolower(argv[i][j]);
            if (ch >= 'a' && ch <= 'z')
                count[ch - 'a']++;       // a=0, b=1, c=2... position
        }
    }

    for (i = 0; i < 26; i++)
        if (count[i] > 0)
            printf("%c : %d\n", 'a' + i, count[i]);
    return 0;
}
Command: prog hello bca
Output:
a : 1
b : 1
c : 1
e : 1
h : 1
l : 2
o : 1

Note: Turbo C me "Arguments" alag set karte hain; modern IDE/terminal me
program ke aage likh do: ./a.out hello bca

Isme kya sikha: argc/argv (command line se input), frequency counting ki technique, aur ch - 'a' wala index trick (character ko array index me badalna).

P13. String Functions (string.h) 📜

#include <stdio.h>
#include <string.h>

int main() {
    char a[50] = "BCA";
    char b[50] = "Students";
    char c[50];

    printf("strlen(a)   = %d\n", strlen(a));        // 3

    strcpy(c, a);
    printf("strcpy c    = %s\n", c);                 // BCA

    strcat(a, " ");
    strcat(a, b);
    printf("strcat a    = %s\n", a);                 // BCA Students

    printf("strcmp(BCA,BCA)  = %d\n", strcmp("BCA", "BCA"));    // 0 = equal
    printf("strcmp(BCA,ABC)  = %d\n", strcmp("BCA", "ABC"));    // +ve: BCA bada

    printf("strrev-like reverse: ");
    for (int i = strlen(c) - 1; i >= 0; i--)   // manual reverse
        printf("%c", c[i]);
    printf("\n");

    printf("strupr/lwr: %s | %s\n", strupr(c), strlwr(c));  // Turbo C
    return 0;
}
strlen(a)   = 3
strcpy c    = BCA
strcat a    = BCA Students
strcmp(BCA,BCA)  = 0
strcmp(BCA,ABC)  = 66   (positive — BCA pehle aata hai dictionary me)
strrev-like reverse: ACB

Isme kya sikha: string.h ke 4 main functions + strcmp ka result number hota hai (0 = equal) — ye sabse zyada poochha jaata hai.

P14. Count Characters in a File 📁

#include <stdio.h>
#include <stdlib.h>

int main() {
    FILE *fp;
    char filename[50], ch;
    long count = 0;

    printf("Enter file name: ");
    scanf("%s", filename);

    fp = fopen(filename, "r");       // r = read mode
    if (fp == NULL) {
        printf("File open nahi hui! (naam/galat path check karo)");
        return 1;
    }

    while ((ch = fgetc(fp)) != EOF)   // ek-ek character padho
        count++;                      // EOF = End Of File

    fclose(fp);                       // file band karna zaroori!
    printf("Total characters = %ld", count);
    return 0;
}
Enter file name: notes.txt
Total characters = 1240

File handling ke 4 steps yaad rakho: 
fopen → check (NULL?) → fgetc/fputc/fprintf in loop → fclose ✔

Isme kya sikha: file handling ka poora cycle — fopen/fgetc/fclose, NULL check, EOF concept. Ye C lab ka sabse important "real programming" program hai.

P16. Copy File — Lowercase → UPPERCASE 🔠

#include <stdio.h>
#include <ctype.h>

int main() {
    FILE *src, *dest;
    char ch;

    src = fopen("input.txt", "r");
    if (src == NULL) {
        printf("Source file nahi mili!");
        return 1;
    }
    dest = fopen("output.txt", "w");

    while ((ch = fgetc(src)) != EOF) {
        if (ch >= 'a' && ch <= 'z')
            ch = toupper(ch);        // small → CAPITAL
        fputc(ch, dest);             // destination me likho
    }

    fclose(src);
    fclose(dest);
    printf("File copy ho gayi (lowercase → UPPERCASE) ✔");
    return 0;
}
input.txt :  "bca first semester" 
output.txt:  "BCA FIRST SEMESTER"

Note: ctype.h ke toupper/tolower hi asli kaam kar rahe hain — 
condition: sirf lowercase letters badlenge, baaki characters (space, ., numbers) same rahenge!

Isme kya sikha: do files ek saath handle karna (read + write), character-by-character processing, toupper ka use. "File copy with transformation" — real utility program! ✅
